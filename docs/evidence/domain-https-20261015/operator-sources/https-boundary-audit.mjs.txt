const hash = data => createHash('sha256').update(data).digest('hex');
async function checkManifest(root, name) {
  const bytes = await readFile(new URL(name, root));
  const manifest = JSON.parse(bytes);
  for (const file of manifest.files) {
    const content = await readFile(new URL(file.path, root));
    assert.equal(hash(content), file.sha256, file.path);
    assert.equal(content.length, file.bytes, file.path);
  }
  return { path: name, sha256: hash(bytes), membersVerified: manifest.files.length };
}

  const response = await new Promise((resolve, reject) => {
    const req = request({ host: '127.0.0.1', port, method, path, headers }, res => {
      let data = ''; res.setEncoding('utf8'); res.on('data', part => { data += part; });
      res.on('end', () => resolve({ status: res.statusCode, headers: res.headers, data }));
    });
    req.on('error', reject); req.end(body);
  });

  assert.equal(response.headers['cache-control'], 'no-store', name);
  assert.equal(response.headers['x-frame-options'], 'DENY', name);
  assert.match(response.headers['content-security-policy'], /connect-src 'self'/, name);

}
