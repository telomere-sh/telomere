def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def run(args):return subprocess.check_output(args,text=True,stderr=subprocess.STDOUT,timeout=30).strip()
def config_snapshot():
    names=[pathlib.Path('/etc/nginx/nginx.conf'),*pathlib.Path('/etc/nginx/sites-enabled').iterdir()]
    return {str(path):sha(path.resolve()) for path in names if path.is_file()}
class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self,*args,**kwargs):return None
opener=urllib.request.build_opener(NoRedirect)
def get(host,path):
    req=urllib.request.Request('http://127.0.0.1'+path,headers={'Host':host})
    try:
        with opener.open(req,timeout=8) as response:return response.status,response.read(),dict(response.headers)
    except urllib.error.HTTPError as error:return error.code,error.read(),dict(error.headers)
