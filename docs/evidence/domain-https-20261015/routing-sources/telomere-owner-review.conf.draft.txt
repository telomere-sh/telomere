# Private owner wallet review helper. Credentials must be generated/stored separately.
server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name review.telomere.sh;

    ssl_certificate /etc/letsencrypt/live/review.telomere.sh/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/review.telomere.sh/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;

    auth_basic "Telomere owner review";
    auth_basic_user_file /etc/nginx/telomere-owner-review.htpasswd;
    client_max_body_size 1k;
    # Avoid retaining private review URLs/query parameters in a shared access log.
    access_log off;
    add_header Cache-Control no-store always;
    add_header Referrer-Policy no-referrer always;
    add_header X-Content-Type-Options nosniff always;
    add_header X-Robots-Tag "noindex, nofollow, noarchive" always;

    location / {
        proxy_pass http://127.0.0.1:3284;
        proxy_http_version 1.1;
        proxy_set_header Host review.telomere.sh;
        proxy_set_header Origin $http_origin;
        proxy_set_header X-Forwarded-Proto https;
        proxy_set_header X-Forwarded-For $remote_addr;
        proxy_set_header X-Real-IP $remote_addr;
        # Consume HTTP Basic authentication at Nginx, not inside the helper.
        proxy_set_header Authorization "";
        proxy_set_header Cookie "";
        proxy_set_header Connection "";
        proxy_cache off;
        proxy_buffering off;
        proxy_redirect off;
        proxy_connect_timeout 5s;
        proxy_send_timeout 15s;
        proxy_read_timeout 60s;
        limit_except GET HEAD POST { deny all; }
    }
}
