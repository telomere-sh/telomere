    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name telomere.sh;

    ssl_certificate /etc/letsencrypt/live/review.telomere.sh/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/review.telomere.sh/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;

    root /var/www/telomere-public/ee6169d1007b11f037fbc409dc1f7575f15117c9;
    index index.html;
    autoindex off;
    disable_symlinks on;

    # Required: generate from the reviewed compiled Website.policyFor(Request('/')).
    # Never install the deliberately invalid placeholder as a live policy.
    include /etc/nginx/snippets/telomere-website-csp.conf;
    add_header X-Content-Type-Options nosniff always;
    add_header Referrer-Policy no-referrer always;
    add_header Cache-Control no-cache always;

    # Match backend namespaces before asset/extension rules; never return the SPA here.
    location ~* ^/(api|v1|health|mcp|x402|console)(/|$) { return 503; }
    location ~* ^/(app\.js|styles\.css)$ { return 503; }

    location ~ (^|/)\. { return 404; }
    location ~* ^/(src|scripts|docs|node_modules|website|dist)(/|$) { return 404; }

    # Missing assets and every path containing a dotted component must remain 404.
    location ~* (^|/)assets(/|$) { try_files $uri =404; }
    location ~ /[^/]*\.[^/]*(/|$) { try_files $uri =404; }

    location = / { try_files /index.html =404; }
    location / { try_files $uri $uri/index.html /page.html; }
}
